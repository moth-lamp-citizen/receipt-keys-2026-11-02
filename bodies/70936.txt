'use strict';
// parseRetryAfter(headerValue, nowMs) -> number | null
// RFC 9110 §10.2.3 (Retry-After). Node 18+, CommonJS, no dependencies. Licence: MIT.
// The clock is never read: nowMs is required and is the only time source.

const MAX_DELTA_SECONDS = 2147483647; // 2^31 - 1; larger values are refused per the order brief

const IMF_FIXDATE = /^(Mon|Tue|Wed|Thu|Fri|Sat|Sun), (\d{2}) (Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec) (\d{4}) (\d{2}):(\d{2}):(\d{2}) GMT$/;
const MONTH = { Jan: 0, Feb: 1, Mar: 2, Apr: 3, May: 4, Jun: 5, Jul: 6, Aug: 7, Sep: 8, Oct: 9, Nov: 10, Dec: 11 };

function parseRetryAfter(headerValue, nowMs) {
  try {
    if (typeof headerValue !== 'string') return null;           // absent, null, undefined, numbers
    if (typeof nowMs !== 'number' || !Number.isFinite(nowMs)) return null;

    const value = headerValue.replace(/^[ \t]+|[ \t]+$/g, ''); // OWS is SP and HTAB only (RFC 9110 §5.6.3)
    if (value === '') return null;

    // delta-seconds: ASCII digits only. "-1", "1.5", "1e3", "5s" are refused here.
    if (/^[0-9]+$/.test(value)) {
      const seconds = Number(value);
      return seconds <= MAX_DELTA_SECONDS ? seconds : null;
    }

    // HTTP-date: IMF-fixdate only. RFC 850 and asctime are refused even though Date.parse takes them.
    const m = IMF_FIXDATE.exec(value);
    if (!m) return null;
    const day = Number(m[2]);
    const month = MONTH[m[3]];
    const year = Number(m[4]);
    const hour = Number(m[5]);
    const minute = Number(m[6]);
    const second = Number(m[7]);
    if (hour > 23 || minute > 59 || second > 59) return null;    // ":60" (leap second) cannot be represented here

    // Date.UTC maps years 0-99 into the 1900s; setUTCFullYear does not.
    const when = new Date(0);
    when.setUTCFullYear(year, month, day);
    when.setUTCHours(hour, minute, second, 0);
    const ms = when.getTime();

    // Reject calendar overflow ("31 Feb" would normalise to March) and any other rewrite.
    if (when.getUTCFullYear() !== year || when.getUTCMonth() !== month || when.getUTCDate() !== day ||
        when.getUTCHours() !== hour || when.getUTCMinutes() !== minute || when.getUTCSeconds() !== second) return null;

    const remainingMs = ms - nowMs;
    if (remainingMs <= 0) return 0;                              // in the past, or exactly now
    const seconds = Math.ceil(remainingMs / 1000);               // whole seconds, rounded up
    return seconds <= MAX_DELTA_SECONDS ? seconds : null;
  } catch (_) {
    return null;                                                 // never throws
  }
}

module.exports = { parseRetryAfter };

=== assumptions and limitations ===
1. The clock is never read: nowMs is required; a non-finite nowMs returns null.
2. delta-seconds are returned as-is; only the HTTP-date form is rounded UP (Math.ceil of the remaining ms).
3. IMF-fixdate is enforced by regex, so RFC 850 and asctime are null even though Date.parse accepts them.
4. Calendar overflow is rejected by round-tripping the parsed components ('31 Feb' -> null, not 3 March).
5. THE DAY-NAME IS NOT CHECKED AGAINST ITS DATE: RFC 9110 gives the grammar, not weekday consistency,
   so 'Tue, 21 Oct 2026 07:28:00 GMT' (a Wednesday) returns a number. One more line refuses it on request.
6. Only OWS (SP and HTAB) is trimmed; a NBSP before the digits makes the value invalid, not 5.
7. ':60' (leap second) is refused: this runtime cannot represent it without normalising to the next minute.
8. Delta-seconds and computed differences above 2^31-1 (2147483647) are null, per the order brief.
9. A non-string headerValue (including 120 as a number) is null, not coerced.
10. Nothing is mutated and the function never throws (its body is inside try/catch).
